        function roughBox(lat, lon) {
          var la = [-90, 90], lo = [-180, 180], h = '', ch = 0, bit = 0, even = true;
          while (h.length < 5) {
            var r = even ? lo : la, v = even ? lon : lat, mid = (r[0] + r[1]) / 2;
            if (v >= mid) { ch = ch * 2 + 1; r[0] = mid; } else { ch = ch * 2; r[1] = mid; }
            even = !even;
            if (++bit === 5) { h += B.charAt(ch); ch = 0; bit = 0; }

    <script nonce="N">
      (async () => {
        const banner = document.getElementById('sponsor-banner');
        if (!banner) return;
        try {
          const params = new URLSearchParams(window.location.search);
          const eventId = params.get('event');
          if (!eventId) return;
          const res = await fetch(
            '/api/v2/events/' + encodeURIComponent(eventId) + '/sponsor-bundle?surface=public'
          );
          const data = await res.json();
          if (!data.ok) return;
          const sponsors = data.bundle?.surface_sponsors?.mobileBanner || [];
          if (!sponsors.length) {
            banner.innerHTML = '<span hidden data-testid="public-sponsor-empty"></span>';
            return;
          }
          const SPONSOR_DEFAULT_LOGO = '/sponsor-default-logo.svg';
          const attr = (v) =>
            String(v == null ? '' : v).replace(/[&<>"']/g, (c) => ({ '&': '&amp;', '<': '&lt;', '>': '&gt;', '"': '&quot;', "'": '&#39;' })[c]);
          banner.innerHTML = sponsors
            .map((s) => {
              // 0083: every sponsor shows an image — its logo through the
              // same-origin proxy (the page CSP is img-src 'self' data:, so an
              // external URL here was always blocked), or the OSE default. The
              // title is display_name → name. attr() escapes quotes too: these
              // values land inside attributes.
              const title = s.title || s.name;
              const src = s.logo_url
                ? '/api/v2/sponsors/' + encodeURIComponent(s.id) + '/logo'
                : SPONSOR_DEFAULT_LOGO;
              const logo =
                '<img src="' + attr(src) + '" alt="' + attr(title) + '" referrerpolicy="no-referrer" data-hide-on-error="1" data-default-src="' + SPONSOR_DEFAULT_LOGO + '" data-testid="public-sponsor-logo">';
              const place = s.address_line
                ? '<small class="sub-line" data-testid="public-sponsor-address">' + attr(s.address_line) + '</small>'
                : '';
              // A tap goes to the https website when there is one, otherwise to
              // the sponsor's own card (title, location, contact). OSE-1438: the
              // resolver sends card_url only when that card renders; without
              // either, the name shows unlinked rather than linking to a 404.
              const link = s.website_url
                ? // OSE-1164: noreferrer (not just noopener) so clicking a sponsor sends NO
                  // Referer header to the sponsor's site — the outbound click leaks neither
                  // the attendee nor which OSE event page they came from.
                  '<a href="' + attr(s.website_url) + '" target="_blank" rel="noopener noreferrer" data-testid="public-sponsor-link"><span data-testid="public-sponsor-name">' + attr(title) + '</span></a>'
                : s.card_url
                  ? '<a href="' + attr(s.card_url) + '" data-testid="public-sponsor-card-link"><span data-testid="public-sponsor-name">' + attr(title) + '</span></a>'
                  : '<span data-testid="public-sponsor-name">' + attr(title) + '</span>';
              return '<div class="sponsor-card" data-testid="public-sponsor-card">' + logo + link + place + '</div>';
            })
            .join('');
          banner.classList.add('sponsor-strip');
          sponsors.forEach((s) => {
            fetch(
              '/api/v2/events/' +
                encodeURIComponent(eventId) +
                '/sponsors/' +
                encodeURIComponent(s.id) +
                '/impression',
              {
                method: 'POST',
                // OSE-1164: the sponsor-signal endpoints are aggregate-only (never-who).
                // Omit credentials so the anonymous-scan cookie (ose_anon) is not even

                // (the identity never reaches the server), not merely a handler that
                // chooses not to read it. Locked by sponsor-signal-never-who.invariant.spec.ts.
                credentials: 'omit',

                // response; without it the write is dropped (honest 202, not counted).
                body: JSON.stringify({ surface: 'public', placement: 'mobileBanner', beacon_token: s.beacon_token }),
              }
            ).catch(() => {});
          });
        } catch {

        }

    <script nonce="N">
      document.addEventListener(
        'error',
        function (ev) {
          var el = ev.target;
          if (el && el.tagName === 'IMG' && el.hasAttribute('data-hide-on-error')) {
            // 0083: a sponsor logo that fails becomes the OSE default instead of
            // vanishing; only if the default itself fails is the image hidden.
            // The fallback is a fixed same-origin path, never read back from
            // the DOM — data-default-src only marks which images may fall back.
            var d = '/sponsor-default-logo.svg';
            if (el.hasAttribute('data-default-src') && el.getAttribute('src') !== d) el.setAttribute('src', d);
            else el.style.display = 'none';
          }
        },
        true
      );

      // "Use my location" (owner 2026-10-10). Phones only, never on its own
      // unless the person already allowed it. The phone turns its position into
      // a 5-character geohash box (about 5 km) here, and only that box is sent,
      // in a form body, to POST /near, which answers with the nearest zip.
      // Nothing is stored. Gate: tests/landing-where.spec.ts.
      (function () {
        var box = document.querySelector('[data-testid="near-me"]');
        var form = document.getElementById('near-form');
        var geo = navigator.geolocation;
        if (!box || !form || !geo || !window.matchMedia('(pointer: coarse)').matches) return;
        var btn = box.querySelector('button');
        var note = box.querySelector('p');
        var B = '0123456789bcdefghjkmnpqrstuvwxyz';
        function roughBox(lat, lon) {
          var la = [-90, 90], lo = [-180, 180], h = '', ch = 0, bit = 0, even = true;
          while (h.length < 5) {
            var r = even ? lo : la, v = even ? lon : lat, mid = (r[0] + r[1]) / 2;
            if (v >= mid) { ch = ch * 2 + 1; r[0] = mid; } else { ch = ch * 2; r[1] = mid; }
            even = !even;
            if (++bit === 5) { h += B.charAt(ch); ch = 0; bit = 0; }
          }
          return h;
        }
        function off() {
          btn.hidden = true;
          note.textContent = 'Location is off. Type your zip below.';
        }
        function find() {
          btn.disabled = true;
          note.textContent = 'Finding your zip…';
          geo.getCurrentPosition(
            function (p) {
              form.elements.g.value = roughBox(p.coords.latitude, p.coords.longitude);
              form.submit();
            },
            function () {
              btn.disabled = false;
              off();
            },
            { enableHighAccuracy: false, maximumAge: 600000, timeout: 15000 }
          );
        }
        btn.addEventListener('click', find);
        var fresh = location.search === '';
        function show(state) {
          if (state === 'denied') return;
          box.hidden = false;
          if (state === 'granted' && fresh) find();
        }
        if (navigator.permissions && navigator.permissions.query) {
          navigator.permissions.query({ name: 'geolocation' }).then(
            function (s) { show(s.state); },
            function () { show('prompt'); }
          );
        } else {
          show('prompt');
        }
      })();
